def sha(data):
    return hashlib.sha256(data).hexdigest()


def json_bytes(value):
    # Adversarial 3.0 must remain a float; canonical JSON intentionally normalizes it.
    return json.dumps(
        value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False
    ).encode("utf-8")


def walk(sample):
    yield sample
    for children in sample["delegates"].values():
        for child in children:
            yield from walk(child)


class Transcript:
    def __init__(self):
        self.count = 0
        self.digest = hashlib.sha256()

    def add(self, *event):
        self.count += 1
        self.digest.update(repr(event).encode("utf-8") + b"\n")

    def result(self):
        return {"events": self.count, "sha256": self.digest.hexdigest()}


class NumericTransform(ast.NodeTransformer):
    """Observe actual results once; retain every original expression and branch."""

    def __init__(self):
        self.path = ()
        self.sites = []
        specs = (
            (("_evaluate",), "set(dimensions)", "live_dimension_keyset"),
            (
                ("_evaluate",),
                "{d.name for d in context.declaration.dimensions}",
                "declared_dimension_set",
            ),
            (
                ("_evaluate",),
                "{bound.metric: bound for bound in context.declaration.bounds}",
                "numeric_bounds_map",
            ),
            (
                ("_evaluate",),
                "any(type(value) is not int or value < 0 for value in dimensions.values())",
                "dimension_values_any",
            ),
            (
                ("_projection_contexts",),
                "{d.name: d.unit for d in declaration.dimensions}",
                "parsed_dimension_map",
            ),
            (
                ("_projection_contexts",),
                "{b.metric: b for b in declaration.bounds}",
                "parsed_bound_map",
            ),
            (
                ("_projection_contexts",),
                "frozenset(dimension_units)",
                "parsed_numeric_dimension_frozenset",
            ),
            (("_validate_sample",), "set(metrics)", "live_metric_keyset"),
            (
                ("_validate_sample",),
                "{b.metric for b in context.declaration.bounds}",
                "declared_metric_set",
            ),
            (
                ("_validate_sample",),
                "any(type(value) is not int or value < 0 for value in metrics.values())",
                "metric_values_any",
            ),
            (
                ("_evaluate_expression", "evaluate"),
                "{name: evaluate(term) for name, term in expression.arguments}",
                "numeric_delegate_arguments",
            ),
            (
                ("validate_execution_samples", "check", "composition"),
                "{name: _evaluate_expression(term, context, row['dimensions']) "
                "for name, term in expression.arguments}",
                "composition_delegate_arguments",
            ),
            (
                ("validate_execution_samples", "check"),
                "canonical_json_bytes(rows[identity])",
                "canonical_bytes",
            ),
        )
        self.patterns = {
            (path, ast.dump(ast.parse(expression, mode="eval").body)): site
            for path, expression, site in specs
        }
        generators = (
            (
                ("_evaluate",),
                "(type(value) is not int or value < 0 for value in dimensions.values())",
                "dimension_values",
            ),
            (
                ("_validate_sample",),
                "(type(value) is not int or value < 0 for value in metrics.values())",
                "metric_values",
            ),
            (
                ("_projection_contexts",),
                "(type(dimension.name) is str for dimension in declaration.dimensions)",
                "dimension_eligibility",
            ),
            (
                ("_projection_contexts",),
                "(type(bound.metric) is str for bound in declaration.bounds)",
                "bound_eligibility",
            ),
        )
        self.generators = {
            (path, ast.dump(ast.parse(expression, mode="eval").body)): site
            for path, expression, site in generators
        }

    def site(self, name, node):
        original = RemoveInstrumentation().visit(deepcopy(node))
        self.sites.append(
            {
                "site": name,
                "function": ".".join(self.path),
                "line": node.lineno,
                "original": ast.unparse(original),
            }
        )

    @staticmethod
    def call(name, args, node):
        return ast.copy_location(
            ast.Call(func=ast.Name(id=name, ctx=ast.Load()), args=args, keywords=[]), node
        )

    def visit_FunctionDef(self, node):
        previous = self.path
        self.path = (*self.path, node.name)
        self.generic_visit(node)
        decorator = None
        if self.path == ("validate_execution_samples", "check"):
            decorator = ast.Name(id=NODE, ctx=ast.Load())
        elif self.path == ("_evaluate_expression", "evaluate"):
            decorator = self.call(
                NUMERIC,
                [ast.Name(id="context", ctx=ast.Load()), ast.Name(id="dimensions", ctx=ast.Load())],
                node,
            )
        elif self.path == ("validate_execution_samples", "check", "composition"):
            decorator = self.call(COMPOSITION, [ast.Name(id="context", ctx=ast.Load())], node)
        if decorator is not None:
            assert not node.decorator_list
            self.site("decorator:" + node.name, node)
            node.decorator_list.append(ast.copy_location(decorator, node))
        self.path = previous
        return node

    def visit(self, node):
        # Save the original shape before instrumenting any child expression.
        key = (self.path, ast.dump(node)) if isinstance(node, ast.expr) else None
        site = self.patterns.get(key)
        result = super().visit(node)
        if site is not None:
            self.site(site, node)
            result = self.call(VALUE, [ast.Constant(value=site), result], node)
        return result

    def visit_GeneratorExp(self, node):
        site = self.generators.get((self.path, ast.dump(node)))
        self.generic_visit(node)
        if site is not None:
            self.site(site + "_generator", node)
            node.elt = self.call(PREDICATE, [ast.Constant(value=site), node.elt], node.elt)
            return self.call(GENERATOR, [ast.Constant(value=site), node], node)
        return node

    def visit_Call(self, node):
        self.generic_visit(node)
        if isinstance(node.func, ast.Name):
            if self.path == ("_projection_contexts",) and node.func.id == "_Context":
                self.site("parsed_context", node)
                return self.call(CONTEXT, [node], node)
            if self.path == ("_read_expression",) and node.func.id == "Expression":
                self.site("parsed_expression", node)
                return self.call(EXPRESSION, [node], node)
        return node

    def visit_Assign(self, node):
        original_value = ast.dump(node.value)
        self.generic_visit(node)
        if self.path == ("_validate_sample",) and original_value == ast.dump(
            ast.parse("_evaluate(context, bound.metric, dimensions)", mode="eval").body
        ):
            self.site("metric_obligation", node)
            node.value = self.call(
                BOUND,
                [
                    node.value,
                    ast.Name(id="context", ctx=ast.Load()),
                    ast.Name(id="bound", ctx=ast.Load()),
                    ast.Name(id="metrics", ctx=ast.Load()),
                ],
                node.value,
            )
        if self.path == ("_evaluate",) and original_value == ast.dump(
            ast.parse("context.numeric_bounds_by_metric", mode="eval").body
        ):
            self.site("numeric_bounds_alias_read", node)
            node.value = self.call(
                VALUE, [ast.Constant(value="numeric_bounds_alias_read"), node.value], node.value
            )
        return node

    def visit_IfExp(self, node):
        selected = self.path == ("_evaluate",) and ast.dump(node.body) == ast.dump(
            ast.parse("context.numeric_dimension_names", mode="eval").body
        )
        self.generic_visit(node)
        if selected:
            self.site("numeric_dimension_names_use", node)
            node.body = self.call(
                VALUE, [ast.Constant(value="numeric_dimension_names_use"), node.body], node.body
            )
        return node


class RemoveInstrumentation(ast.NodeTransformer):
    def visit_FunctionDef(self, node):
        self.generic_visit(node)
        node.decorator_list = [
            item
            for item in node.decorator_list
            if not (isinstance(item, ast.Name) and item.id == NODE)
            and not (
                isinstance(item, ast.Call)
                and isinstance(item.func, ast.Name)
                and item.func.id in (NUMERIC, COMPOSITION)
            )
        ]
        return node

    def visit_Call(self, node):
        self.generic_visit(node)
        if isinstance(node.func, ast.Name):
            if node.func.id in (VALUE, PREDICATE, GENERATOR):
                return node.args[1]
            if node.func.id in (CONTEXT, EXPRESSION, BOUND):
                return node.args[0]
        return node


class Diagnostic:
    """Diagnostic-only counters. No callback changes a value or acceptance decision."""

    def __init__(self, node_paths):
        self.node_paths = node_paths
        self.current_node = None
        self.current_group = None
        self.active_compile = None
        self.compilations = []
        self.contexts = Counter()
        self.nodes = Counter()
        self.units = Counter()
        self.unit_edges = Counter()
        self.numeric_edges = Counter()
        self.composition_edges = Counter()
        self.numeric_metrics = Counter()
        self.numeric_operators = Counter()
        self.parsed_operators = Counter()
        self.expression_tokens = {}
        self.reader_receipts = []
        self.counts = Counter()
        self.metadata = Counter()
        self.predicates = {
            name: {"generators": 0, "inspections": 0, "true": 0, "false": 0}
            for name in (
                "dimension_values",
                "metric_values",
                "dimension_eligibility",
                "bound_eligibility",
            )
        }
        self.constructors = {site: {"builds": 0, "entries": 0} for site in CONSTRUCTORS}
        self.streams = {
            name: Transcript()
            for name in (
                "parsed_expressions",
                "unit_entries",
                "unit_returns",
                "node_entries",
                "node_returns",
                "metric_obligations",
                "numeric_entries",
                "numeric_returns",
                "numeric_sessions",
                "expression_entries",
                "expression_returns",
                "composition_entries",
                "composition_returns",
                "canonical_calls",
            )
        }
        self.numeric_depth = 0

    def value(self, site, actual):
        if site in CONSTRUCTORS:
            assert type(actual) is CONSTRUCTORS[site]
            self.constructors[site]["builds"] += 1
            self.constructors[site]["entries"] += len(actual)
        elif site in ("dimension_values_any", "metric_values_any"):
            assert type(actual) is bool
            self.counts[site + "_calls"] += 1
            self.counts[site + ("_true" if actual else "_false_exhaustions")] += 1
        elif site == "canonical_bytes":
            assert type(actual) is bytes and self.current_node is not None
            self.counts["canonical_json_calls"] += 1
            self.counts["canonical_json_bytes_returned"] += len(actual)
            self.streams["canonical_calls"].add(self.current_node, len(actual))
        elif site == "numeric_bounds_alias_read":
            self.metadata[site] += 1
            self.metadata[
                "numeric_bounds_alias_uses" if actual is not None else "numeric_bounds_alias_none"
            ] += 1
        elif site == "numeric_dimension_names_use":
            assert type(actual) is frozenset
            self.metadata[site] += 1
        else:
            raise AssertionError(site)
        return actual

    def predicate(self, site, actual):
        assert type(actual) is bool
        row = self.predicates[site]
        row["inspections"] += 1
        row["true" if actual else "false"] += 1
        return actual

    def generator(self, site, actual):
        self.predicates[site]["generators"] += 1
        return actual

    def context(self, actual):
        assert self.active_compile is not None
        declaration = actual.declaration
        self.contexts[declaration.id] += 1
        self.counts["context_dimension_entries"] += len(declaration.dimensions)
        self.counts["context_bound_entries"] += len(declaration.bounds)
        self.compilations[self.active_compile]["contexts"] += 1
        names = getattr(actual, "numeric_dimension_names", None)
        bounds = getattr(actual, "numeric_bounds_by_metric", None)
        if names is not None:
            assert type(names) is frozenset and len(names) == len(actual.dimension_units)
            self.metadata["contexts_with_numeric_names"] += 1
            self.metadata["numeric_name_entries_retained"] += len(names)
        if bounds is not None:
            assert bounds is actual.bounds_by_metric
            self.metadata["contexts_with_numeric_bound_alias"] += 1
        return actual

    def expression(self, actual):
        assert self.active_compile is not None
        token = self.counts["parsed_expressions"]
        self.counts["parsed_expressions"] += 1
        self.expression_tokens[id(actual)] = token
        self.parsed_operators[actual.op] += 1
        self.streams["parsed_expressions"].add(
            self.active_compile,
            token,
            actual.op,
            actual.name,
            actual.metric,
            actual.value,
            actual.unit,
            tuple(self.expression_tokens[id(term)] for term in actual.terms),
            tuple((name, self.expression_tokens[id(term)]) for name, term in actual.arguments),
        )
        return actual

    def bound(self, actual, context, bound, metrics):
        self.counts["metric_obligations"] += 1
        self.streams["metric_obligations"].add(
            self.current_node,
            context.declaration.id,
            bound.metric,
            bound.relation,
            metrics[bound.metric],
            actual,
        )
        return actual

    def node(self, original):
        def check(identity, sample, depth, budget):
            path = self.node_paths[id(sample)]
            previous, self.current_node = self.current_node, path
            self.streams["node_entries"].add(path, identity, depth, budget[0])
            try:
                result = original(identity, sample, depth, budget)
                self.streams["node_returns"].add(path, identity, depth, budget[0], result)
                return result
            finally:
                self.current_node = previous

        return check

    def numeric(self, context, dimensions):
        session = self.counts["numeric_expression_sessions"]
        self.counts["numeric_expression_sessions"] += 1
        self.streams["numeric_sessions"].add(
            session,
            self.current_node,
            context.declaration.id,
            tuple(dimensions.items()),
        )

        def decorate(original):
            depth = 0

            def evaluate(expression):
                nonlocal depth
                token = self.expression_tokens[id(expression)]
                event = (session, depth, token)
                self.counts["numeric_expression_calls"] += 1
                self.numeric_operators[expression.op] += 1
                self.streams["expression_entries"].add(*event)
                if expression.op == "delegate":
                    child = context.delegates[expression.name]
                    self.numeric_edges[
                        (
                            context.declaration.id,
                            expression.name,
                            child.declaration.id,
                            expression.metric,
                        )
                    ] += 1
                depth += 1
                try:
                    result = original(expression)
                    self.counts["numeric_expression_returns"] += 1
                    self.streams["expression_returns"].add(*event, result)
                    return result
                finally:
                    depth -= 1

            return evaluate

        return decorate

    def composition(self, context):
        def decorate(original):
            depth = 0

            def compose(expression):
                nonlocal depth
                event = (self.current_node, depth, self.expression_tokens[id(expression)])
                self.counts["composition_calls"] += 1
                self.streams["composition_entries"].add(*event)
                if expression.op == "delegate":
                    child = context.delegates[expression.name]
                    self.composition_edges[
                        (
                            context.declaration.id,
                            expression.name,
                            child.declaration.id,
                            expression.metric,
                        )
                    ] += 1
                depth += 1
                try:
                    result = original(expression)
                    self.counts["composition_returns"] += 1
                    self.streams["composition_returns"].add(*event, result)
                    return result
                finally:
                    depth -= 1

            return compose

        return decorate

    def attach(self, module):
        original_compile, original_unit = module._projection_contexts, module._expression_unit
        original_node, original_reader = module._validate_sample, module.validate_execution_samples
        original_evaluate = module._evaluate

        def compile_projection(projection):
            assert self.active_compile is None
            self.active_compile = len(self.compilations)
            self.compilations.append(
                {"declarations": len(projection["declarations"]), "contexts": 0}
            )
            try:
                result = original_compile(projection)
                row = self.compilations[self.active_compile]
                assert row["contexts"] == row["declarations"] == len(result)
                return result
            finally:
                self.active_compile = None

        def expression_unit(expression, context, stack, relation):
            assert self.active_compile is not None
            identity = context.declaration.id
            event = (
                self.active_compile,
                identity,
                stack,
                relation,
                self.expression_tokens[id(expression)],
            )
            self.counts["expression_unit_calls"] += 1
            self.units[identity] += 1
            self.streams["unit_entries"].add(*event)
            if expression.op == "delegate":
                child = context.delegates[expression.name]
                self.unit_edges[
                    (identity, expression.name, child.declaration.id, expression.metric, relation)
                ] += 1
            result = original_unit(expression, context, stack, relation)
            self.counts["expression_unit_returns"] += 1
            self.streams["unit_returns"].add(*event, result)
            return result

        def evaluate(context, metric, dimensions):
            event = (
                self.current_node,
                self.numeric_depth,
                context.declaration.id,
                metric,
                tuple(dimensions.items()),
            )
            self.counts["numeric_evaluate_calls"] += 1
            self.counts["numeric_evaluate_nonrecursive_calls"] += self.numeric_depth == 0
            self.numeric_metrics[(context.declaration.id, metric)] += 1
            self.streams["numeric_entries"].add(*event)
            self.numeric_depth += 1
            try:
                result = original_evaluate(context, metric, dimensions)
                self.counts["numeric_evaluate_returns"] += 1
                self.streams["numeric_returns"].add(*event, result)
                return result
            finally:
                self.numeric_depth -= 1

        def validate_node(context, dimensions, metrics):
            result = original_node(context, dimensions, metrics)
            self.nodes[context.declaration.id] += 1
            return result

        def reader(projection, declaration, samples):
            assert self.current_group is None
            self.current_group = len(self.reader_receipts)
            try:
                result = original_reader(projection, declaration, samples)
                self.reader_receipts.append(
                    (declaration, len(projection["declarations"]), tuple(samples))
                )
                return result
            finally:
                self.current_group = None

        module._projection_contexts = compile_projection
        module._expression_unit = expression_unit
        module._evaluate = evaluate
        module._validate_sample = validate_node
        module.validate_execution_samples = reader

    def result(self):
        assert self.numeric_depth == 0 and self.current_node is None
        assert self.active_compile is None and self.current_group is None
        for name in ("expression_unit", "numeric_evaluate", "numeric_expression", "composition"):
            assert self.counts[name + "_calls"] == self.counts[name + "_returns"]

        def edges(counter, *, relation=False):
            return [
                {
                    "parent": key[0],
                    "tag": key[1],
                    "child": key[2],
                    "metric": key[3],
                    **({"relation": key[4]} if relation else {}),
                    "visits": count,
                }
                for key, count in sorted(counter.items())
            ]

        required = {
            **dict(self.counts),
            "compilation_calls": len(self.compilations),
            "compilations": self.compilations,
            "parsed_contexts": sum(self.contexts.values()),
            "contexts_by_declaration": dict(self.contexts),
            "parsed_expression_operators": dict(self.parsed_operators),
            "unit_visits_by_declaration": dict(self.units),
            "unit_delegate_edges": edges(self.unit_edges, relation=True),
            "numeric_delegate_edges": edges(self.numeric_edges),
            "composition_delegate_edges": edges(self.composition_edges),
            "numeric_calls_by_metric": [
                {"declaration": key[0], "metric": key[1], "calls": value}
                for key, value in sorted(self.numeric_metrics.items())
            ],
            "numeric_expression_operators": dict(self.numeric_operators),
            "public_reader_calls": len(self.reader_receipts),
            "public_reader_root_occurrences": sum(len(row[2]) for row in self.reader_receipts),
            "successful_node_checks": sum(self.nodes.values()),
            "nodes_by_declaration": dict(self.nodes),
            "constructors": {
                site: value
                for site, value in self.constructors.items()
                if site not in CHANGED_CONSTRUCTORS
            },
            "live_predicates": {
                site: self.predicates[site] for site in ("dimension_values", "metric_values")
            },
            "ordered_transcripts": {name: value.result() for name, value in self.streams.items()},
        }
        metadata = {
            "constructors": {
                site: value
                for site, value in self.constructors.items()
                if site in CHANGED_CONSTRUCTORS
            },
            "eligibility_predicates": {
                site: self.predicates[site]
                for site in ("dimension_eligibility", "bound_eligibility")
            },
            "counts": dict(self.metadata),
        }
        return {"required_work": required, "numeric_metadata": metadata}


def make_module(name, source, filename, *, diagnostic=None, proposal=False):
    module = ModuleType(name)
    module.__file__ = str(filename)
    module.__package__ = ""
    sys.modules[name] = module
    manifest = None
    if diagnostic is None:
        code = compile(source, str(filename), "exec")
    else:
        original = ast.parse(source, filename=str(filename))
        assert all(
            not isinstance(node, ast.Name) or node.id not in HELPERS for node in ast.walk(original)
        )
        transformer = NumericTransform()
        transformed = transformer.visit(deepcopy(original))
        ast.fix_missing_locations(transformed)
        restored = RemoveInstrumentation().visit(deepcopy(transformed))
        assert ast.dump(restored) == ast.dump(original)
        expected = Counter(
            {site: 1 for site in CONSTRUCTORS if site != "parsed_numeric_dimension_frozenset"}
        )
        expected.update(
            {
                "dimension_values_any": 1,
                "metric_values_any": 1,
                "canonical_bytes": 1,
                "dimension_values_generator": 1,
                "metric_values_generator": 1,
                "parsed_context": 1,
                "parsed_expression": 4,
                "metric_obligation": 1,
                "decorator:check": 1,
                "decorator:evaluate": 1,
                "decorator:composition": 1,
            }
        )
        if proposal:
            expected.update(
                {
                    "parsed_numeric_dimension_frozenset": 1,
                    "dimension_eligibility_generator": 1,
                    "bound_eligibility_generator": 1,
                    "numeric_bounds_alias_read": 1,
                    "numeric_dimension_names_use": 1,
                }
            )
        assert Counter(site["site"] for site in transformer.sites) == expected
        callbacks = {
            VALUE: diagnostic.value,
            CONTEXT: diagnostic.context,
            EXPRESSION: diagnostic.expression,
            PREDICATE: diagnostic.predicate,
            GENERATOR: diagnostic.generator,
            BOUND: diagnostic.bound,
            NODE: diagnostic.node,
            NUMERIC: diagnostic.numeric,
            COMPOSITION: diagnostic.composition,
        }
        module.__dict__.update(callbacks)
        code = compile(transformed, str(filename), "exec")
        instrumented = ast.unparse(transformed) + "\n"
        manifest = {
            "sites": transformer.sites,
            "inverse_transform_matches_original_ast": True,
            "instrumented_source": instrumented,
            "instrumented_source_sha256": sha(instrumented.encode()),
        }
    exec(code, module.__dict__)
    assert Path(module.__file__).resolve() == Path(filename).resolve()
    if diagnostic is not None:
        diagnostic.attach(module)
    return module, manifest

                    actual is supplied
                    for actual, supplied in zip(observations, original[2], strict=True)
                )
                receipts.append(
                    {
                        "group": index,
                        "declaration": declaration,
                        "projection_declarations": size,
                        "samples": groups[index]["samples"],
                    }
                )
            answer["public_reader_receipts"] = receipts
            answer["verification_status"] = "passed"
            return answer

        with tempfile.TemporaryDirectory(prefix="numeric-reader-profile-") as directory:
            scratch = Path(directory)
            assert not scratch.is_relative_to(checkout)
            (scratch / "scope.json").write_bytes(json_bytes({"groups": 46, "product_fixtures": 0}))
            report["cleanup"].update(
                profile_scratch_created=True, profile_scratch_path=str(scratch)
            )
            for diagnostic in (True, False):
                for label in ("baseline-reader", "proposed-reader"):
                    name = ("diagnostic:" if diagnostic else "elapsed:") + label
                    print("begin", name, flush=True)
                    report["measurements"][name] = run_workload(label, diagnostic)
                    write("run.json", report)
                    print("completed", name, flush=True)
                if diagnostic:
                    baseline = report["measurements"]["diagnostic:baseline-reader"]
                    proposed = report["measurements"]["diagnostic:proposed-reader"]
                    assert (
                        baseline["counts"]["required_work"] == proposed["counts"]["required_work"]
                    )
                    assert baseline["public_reader_receipts"] == proposed["public_reader_receipts"]
                    report["diagnostic_equivalence"] = {
                        "all_required_counts_equal": True,
                        "all_ordered_transcripts_equal": True,
                        "all_unit_numeric_and_composition_edge_counts_equal": True,
                        "exact_groups_root_tokens_and_total_node_inventory_equal": True,
                        "changed_accounting_only": (
                            "Numeric declaration metadata construction, guarded eligibility "
                            "checks, and reuse of parsed metadata references."
                        ),
                    }
                    write("run.json", report)
        report["cleanup"]["profile_scratch_removed"] = not scratch.exists()
        report["cleanup"]["temporary_root_finally_empty"] = not tuple(temporary_root.iterdir())
        assert (
            report["cleanup"]["profile_scratch_removed"]
            and report["cleanup"]["temporary_root_finally_empty"]
        )
        for label, module in {**plain_modules, "installed-reader": installed_reader}.items():
            assert all(
                getattr(module, name) is original
                for name, original in original_functions[label].items()
            )
        assert canonical_owner.canonical_json_bytes is canonical_json_bytes
        assert identity_owner.source_identity is source_identity
        assert all(sha(path.read_bytes()) == digest for path, digest in pinned_files.items())
        report["unmodified_modules_and_pinned_files_preserved"] = True
        report["state"] = "completed"
        status = 0
    except BaseException as error:
        report["state"] = "failed"
        report["error"] = {"type": type(error).__name__, "message": str(error)}
        traceback.print_exc()
    finally:
        if scratch is not None:
            report["cleanup"]["profile_scratch_removed"] = not scratch.exists()
        if os.environ.get("TMPDIR") and Path(os.environ["TMPDIR"]).is_dir():
            report["cleanup"]["temporary_root_finally_empty"] = not tuple(
                Path(os.environ["TMPDIR"]).iterdir()
            )
        report["source_end"] = source_identity(checkout)
        if report["source_end"] != report["source_start"]:
            report["state"] = "source-changed"
            status = 1
        report["pinned_files_unchanged_at_exit"] = all(
            path.exists() and sha(path.read_bytes()) == digest
            for path, digest in pinned_files.items()
        )
        if not report["pinned_files_unchanged_at_exit"]:
            report["state"] = "pinned-file-changed"
            status = 1
        report["exit_status"] = status
        write("run.json", report)
        print(
            "original child final status",
            status,
            "source",
            report["source_end"],
            "cleanup",
            report["cleanup"],
            flush=True,
        )
    return status
