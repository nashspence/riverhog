class ExclusiveTimers:
    """Attribute each instrumented interval to only the top stack category."""

    def __init__(self):
        self.totals = {name: [0, 0] for name in CATEGORIES}
        self.stack = []
        self.previous = None
        self.wall_clock, self.cpu_clock = time.perf_counter_ns, time.process_time_ns

    def tick(self):
        current = self.wall_clock(), self.cpu_clock()
        if self.previous is not None and self.stack:
            total = self.totals[self.stack[-1]]
            total[0] += current[0] - self.previous[0]
            total[1] += current[1] - self.previous[1]
        self.previous = current

    @contextmanager
    def scope(self, name):
        self.tick()
        self.stack.append(name)
        try:
            yield
        finally:
            self.tick()
            assert self.stack.pop() == name

    def report(self):
        assert not self.stack
        return {
            name: {"wall_ns": value[0], "cpu_ns": value[1]}
            for name, value in self.totals.items()
        }


def walk(sample):
    yield sample
    for children in sample["delegates"].values():
        for child in children:
            yield from walk(child)


def operation_name(operation):
    return operation.__module__ + "." + operation.__qualname__


def build_actions(events):
    """Collapse nested capture events already performed by real helper calls."""
    assert [event["event"] for event in events] == list(range(len(events)))
    ends = {
        (event["kind"], event["original_event"]): event
        for event in events
        if "original_event" in event
    }
    actions, constructor_projects = [], []
    position = 0
    native_prefix = "tests.unit.test_upload_journal_"
    native_modules = {
        native_prefix + "owned_source_growth",
        native_prefix + "validation_open_growth",
        native_prefix + "validation_advance_growth",
    }
    while position < len(events):
        event = events[position]
        kind = event["kind"]
        if kind == "projection-begin" and event["module"] == native_prefix + "owned_source_growth":
            assert not constructor_projects
            start = position
            while position < len(events):
                current = events[position]
                if current["kind"] != "projection-begin" or current["module"] not in native_modules:
                    break
                constructor_projects.append(current)
                position = ends["projection-return", position]["event"] + 1
            actions.append({"kind": "native-constructor", "event": start})
            continue
        if kind == "emission-begin":
            end = ends["emission-return", position]
            actions.append({**event, "sample": end["sample"]})
            position = end["event"] + 1
            continue
        if kind == "observer-construction-begin":
            actions.append(event)
            position = ends["observer-ready", position]["event"] + 1
            continue
        if kind == "observer-context-closed":
            actions.append(event)
        elif kind == "observer-completion-begin":
            end = ends["observer-completion-return", position]
            actions.append({**event, "sample": end["sample"]})
            position = end["event"] + 1
            continue
        elif kind == "validation-begin":
            if event["module"] != "tests.support.growth_batch":
                assert event["module"] in native_modules, event
                actions.append(event)
            position = ends["validation-return", position]["event"] + 1
            continue
        elif kind == "projection-begin":
            if event["module"] == "tests.support.growth_batch":
                assert position == 0
            else:
                assert event["module"] == native_prefix + "validation_open_growth", event
                actions.append({**event, "kind": "opening-projection"})
            position = ends["projection-return", position]["event"] + 1
            continue
        else:
            assert kind in {"caller-turn-begin", "caller-turn-return", "observer-body-exit"}, event
        position += 1
    assert len(constructor_projects) == 46
    assert Counter(row["kind"] for row in actions) == {
        "native-constructor": 1,
        "emission-begin": 16,
        "observer-construction-begin": 8,
        "observer-context-closed": 8,
        "observer-completion-begin": 8,
        "validation-begin": 106,
        "opening-projection": 1,
    }
    return actions, constructor_projects

        @contextmanager
        def instrument(timers, counts, reader_receipts, projection_receipts):
            category_compile, category_source, category_samples = CATEGORIES[:3]

            def categorized(category, function, counter=None):
                @wraps(function)
                def invoke(*values, **options):
                    with timers.scope(category):
                        if counter is not None:
                            counts[counter] += 1
                        return function(*values, **options)
                return invoke

            def project(bindings):
                with timers.scope(category_compile):
                    counts["projection_calls"] += 1
                    projection_receipts.append(tuple(binding.operation for binding in bindings))
                    return originals["project"](bindings)

            def compile_projection(projection):
                with timers.scope(category_compile):
                    counts["compilation_calls"] += 1
                    counts["compiled_declaration_occurrences"] += len(projection["declarations"])
                    return originals["compile"](projection)

            def read(projection, identity, samples):
                with timers.scope(category_samples):
                    counts["public_reader_calls"] += 1
                    result = originals["read"](projection, identity, samples)
                    reader_receipts.append((identity, len(projection["declarations"]), tuple(samples)))
                    return result

            def node(context, dimensions, metrics):
                result = originals["node"](context, dimensions, metrics)
                counts["validated_node_occurrences"] += 1
                counts["node:" + context.declaration.id] += 1
                return result

            def expression_unit(*values, **options):
                counts["expression_unit_traversals"] += 1
                return originals["units"](*values, **options)

            class TimedHash:
                def __init__(self, value):
                    self.value = value

                def __getattr__(self, name):
                    return getattr(self.value, name)

                def digest(self):
                    with timers.scope(category_source):
                        return self.value.digest()

                def hexdigest(self):
                    with timers.scope(category_source):
                        return self.value.hexdigest()

                def update(self, data):
                    with timers.scope(category_source):
                        return self.value.update(data)

                def copy(self):
                    with timers.scope(category_source):
                        return TimedHash(self.value.copy())

            def hash_value(*values, **options):
                with timers.scope(category_source):
                    counts["sha256_constructions"] += 1
                    return TimedHash(originals["hash"](*values, **options))

                result["nodes_by_declaration"] = dict(checked_nodes)
                result["exclusive_categories"] = timers.report()
                result["public_reader_receipts"] = [
                    {"declaration": declaration, "projection_declarations": size,
                     "samples": [tokens[id(value)] for value in samples]}
                    for declaration, size, samples in reader_receipts
                ]
            return result

        write("run.json", report)
        with tempfile.TemporaryDirectory(prefix="observer-lifetime-replay-") as temporary:
            scratch = Path(temporary)
            assert not scratch.is_relative_to(checkout)
            report["cleanup"].update(profile_scratch_created=True, profile_scratch_path=str(scratch))
            for diagnostic in (True, False):
                for route in ("legacy", "extended"):
                    name = ("instrumented:" if diagnostic else "elapsed:") + route
                    print("begin", name, "literal retained observer work; no product fixture", flush=True)
                    result = replay_route(route, diagnostic)
                    report["measurements"][name] = result
                    write("run.json", report)
                    print("completed", name, "wall_ns", result["wall_ns"], "cpu_ns", result["cpu_ns"], flush=True)
        report["cleanup"]["profile_scratch_removed"] = not scratch.exists()
        assert report["cleanup"]["profile_scratch_removed"]
        report["state"] = "completed"
        status = 0
    except BaseException as error:
        report["state"] = "failed"
        report["error"] = {"type": type(error).__name__, "message": str(error)}
        traceback.print_exc()
    finally:
        if scratch is not None:
            report["cleanup"]["profile_scratch_removed"] = not scratch.exists()
        report["source_end"] = source_identity(checkout)
        if report["source_end"] != report["source"]:
            report["state"] = "source-changed"
            status = 1
        report["exit_status"] = status
        write("run.json", report)
        print("original child final status", status, "cleanup", json.dumps(report["cleanup"], sort_keys=True), flush=True)
    return status
